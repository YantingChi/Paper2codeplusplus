"""model.py

This module defines the TransformerModel class and all of its internal sub-modules,
which together implement the Transformer architecture as described in "Attention Is All You Need."

The implementation follows a modular design using PyTorch. All configuration parameters are
provided via a parameters dictionary (typically derived from config.yaml), with sensible default values.
This file includes the definitions for the following components:
    - PositionalEncoding (sinusoidal or learned)
    - MultiHeadAttention (with scaled dot-product attention)
    - FeedForward network (position-wise, with ReLU activation)
    - EncoderLayer and DecoderLayer (with residual connections and layer normalization)
    - TransformerModel, which encapsulates the embedding layers (with weight tying), encoder, decoder,
      and generator (final linear projection). It also provides a "generate" method for beam search decoding.

Required packages:
    torch==1.9.0
    numpy==1.21.0
"""

import math
from typing import Any, Dict, Optional

import torch
import torch.nn as nn
import torch.nn.functional as F


class PositionalEncoding(nn.Module):
    """
    Implements positional encoding as described in "Attention Is All You Need".
    Supports fixed sinusoidal encoding (default) and an option for learned positional embeddings.
    """
    def __init__(self, d_model: int, dropout: float = 0.1, max_len: int = 5000, mode: str = "sinusoidal") -> None:
        super(PositionalEncoding, self).__init__()
        self.dropout = nn.Dropout(dropout)
        if mode == "sinusoidal":
            pe = torch.zeros(max_len, d_model)
            position = torch.arange(0, max_len, dtype=torch.float).unsqueeze(1)
            div_term = torch.exp(torch.arange(0, d_model, 2).float() * -(math.log(10000.0) / d_model))
            pe[:, 0::2] = torch.sin(position * div_term)
            pe[:, 1::2] = torch.cos(position * div_term)
            pe = pe.unsqueeze(0)  # shape: (1, max_len, d_model)
            self.register_buffer("pe", pe)
            self.mode = "sinusoidal"
        elif mode == "learned":
            self.pe = nn.Embedding(max_len, d_model)
            self.mode = "learned"
        else:
            raise ValueError(f"Unknown positional encoding mode: {mode}")

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: Tensor of shape (batch_size, seq_len, d_model)
        Returns:
            Tensor of the same shape as x with positional encodings added (after dropout).
        """
        seq_len = x.size(1)
        if self.mode == "sinusoidal":
            x = x + self.pe[:, :seq_len, :]
        else:  # learned positional encoding
            positions = torch.arange(0, seq_len, device=x.device).unsqueeze(0).expand(x.size(0), seq_len)
            x = x + self.pe(positions)
        return self.dropout(x)


class MultiHeadAttention(nn.Module):
    """
    Implements scaled dot-product multi-head attention.
    Projects inputs to multiple heads, computes attention weights with scaling,
    and concatenates the outputs before a final linear projection.
    """
    def __init__(self, d_model: int, num_heads: int, d_k: int, d_v: int, dropout: float = 0.1) -> None:
        super(MultiHeadAttention, self).__init__()
        self.d_model = d_model
        self.num_heads = num_heads
        self.d_k = d_k
        self.d_v = d_v
        self.linear_q = nn.Linear(d_model, num_heads * d_k)
        self.linear_k = nn.Linear(d_model, num_heads * d_k)
        self.linear_v = nn.Linear(d_model, num_heads * d_v)
        self.linear_out = nn.Linear(num_heads * d_v, d_model)
        self.dropout = nn.Dropout(dropout)
        self.scale = math.sqrt(d_k)

    def forward(self, query: torch.Tensor, key: torch.Tensor, value: torch.Tensor,
                mask: Optional[torch.Tensor] = None) -> torch.Tensor:
        """
        Args:
            query: Tensor of shape (batch_size, seq_len_q, d_model)
            key:   Tensor of shape (batch_size, seq_len_k, d_model)
            value: Tensor of shape (batch_size, seq_len_v, d_model)
            mask:  Optional tensor that prevents attention to certain positions.
                   Expected shape is broadcastable to (batch_size, num_heads, seq_len_q, seq_len_k)
        Returns:
            Tensor of shape (batch_size, seq_len_q, d_model)
        """
        batch_size = query.size(0)
        Q = self.linear_q(query)  # (batch_size, seq_len_q, num_heads*d_k)
        K = self.linear_k(key)    # (batch_size, seq_len_k, num_heads*d_k)
        V = self.linear_v(value)  # (batch_size, seq_len_v, num_heads*d_v)

        # Reshape and transpose for multi-head attention: (batch_size, num_heads, seq_len, d_k/d_v)
        Q = Q.view(batch_size, -1, self.num_heads, self.d_k).transpose(1, 2)
        K = K.view(batch_size, -1, self.num_heads, self.d_k).transpose(1, 2)
        V = V.view(batch_size, -1, self.num_heads, self.d_v).transpose(1, 2)

        # Compute scaled dot-product attention scores
        scores = torch.matmul(Q, K.transpose(-2, -1)) / self.scale  # (batch_size, num_heads, seq_len_q, seq_len_k)
        if mask is not None:
            scores = scores.masked_fill(mask, float('-inf'))
        attn = torch.softmax(scores, dim=-1)
        attn = self.dropout(attn)
        output = torch.matmul(attn, V)  # (batch_size, num_heads, seq_len_q, d_v)
        # Concatenate heads and project
        output = output.transpose(1, 2).contiguous().view(batch_size, -1, self.num_heads * self.d_v)
        output = self.linear_out(output)
        return output


class FeedForward(nn.Module):
    """
    Implements the position-wise feed-forward network that consists of
    two linear transformations with a ReLU activation in between.
    """
    def __init__(self, d_model: int, d_ff: int, dropout: float = 0.1) -> None:
        super(FeedForward, self).__init__()
        self.linear1 = nn.Linear(d_model, d_ff)
        self.linear2 = nn.Linear(d_ff, d_model)
        self.dropout = nn.Dropout(dropout)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = self.linear1(x)
        x = F.relu(x)
        x = self.dropout(x)
        x = self.linear2(x)
        return x


class EncoderLayer(nn.Module):
    """
    Implements a single layer of the Transformer encoder which includes:
      - Multi-head self-attention with residual connection and layer normalization.
      - Position-wise feed-forward network with residual connection and layer normalization.
    """
    def __init__(self, d_model: int, num_heads: int, d_ff: int, d_k: int, d_v: int, dropout: float = 0.1) -> None:
        super(EncoderLayer, self).__init__()
        self.self_attn = MultiHeadAttention(d_model, num_heads, d_k, d_v, dropout)
        self.feed_forward = FeedForward(d_model, d_ff, dropout)
        self.norm1 = nn.LayerNorm(d_model)
        self.norm2 = nn.LayerNorm(d_model)
        self.dropout = nn.Dropout(dropout)

    def forward(self, x: torch.Tensor, mask: Optional[torch.Tensor] = None) -> torch.Tensor:
        attn_output = self.self_attn(x, x, x, mask)
        x = self.norm1(x + self.dropout(attn_output))
        ff_output = self.feed_forward(x)
        x = self.norm2(x + self.dropout(ff_output))
        return x


class DecoderLayer(nn.Module):
    """
    Implements a single layer of the Transformer decoder which includes:
      - Masked multi-head self-attention with causal masking.
      - Encoder-decoder (cross) attention with residual connection and layer normalization.
      - Position-wise feed-forward network with residual connection and layer normalization.
    """
    def __init__(self, d_model: int, num_heads: int, d_ff: int, d_k: int, d_v: int, dropout: float = 0.1) -> None:
        super(DecoderLayer, self).__init__()
        self.self_attn = MultiHeadAttention(d_model, num_heads, d_k, d_v, dropout)
        self.cross_attn = MultiHeadAttention(d_model, num_heads, d_k, d_v, dropout)
        self.feed_forward = FeedForward(d_model, d_ff, dropout)
        self.norm1 = nn.LayerNorm(d_model)
        self.norm2 = nn.LayerNorm(d_model)
        self.norm3 = nn.LayerNorm(d_model)
        self.dropout = nn.Dropout(dropout)

    def forward(self, x: torch.Tensor, encoder_output: torch.Tensor,
                tgt_mask: Optional[torch.Tensor] = None) -> torch.Tensor:
        self_attn_output = self.self_attn(x, x, x, tgt_mask)
        x = self.norm1(x + self.dropout(self_attn_output))
        cross_attn_output = self.cross_attn(x, encoder_output, encoder_output, mask=None)
        x = self.norm2(x + self.dropout(cross_attn_output))
        ff_output = self.feed_forward(x)
        x = self.norm3(x + self.dropout(ff_output))
        return x


class TransformerModel(nn.Module):
    """
    The TransformerModel encapsulates the complete Transformer architecture.
    It includes the embedding layer (with scaling), positional encoding, stacked encoder and decoder,
    and a generator for producing logits over the vocabulary. It also supports beam search decoding via the
    generate() method.
    """
    def __init__(self, params: Dict[str, Any]) -> None:
        super(TransformerModel, self).__init__()
        # Configuration parameters with default values as specified in config.yaml
        self.d_model: int = params.get("d_model", 512)
        self.d_ff: int = params.get("d_ff", 2048)
        self.num_heads: int = params.get("num_heads", 8)
        self.encoder_layers: int = params.get("encoder_layers", 6)
        self.decoder_layers: int = params.get("decoder_layers", 6)
        self.dropout_prob: float = params.get("dropout", 0.1)
        self.d_k: int = params.get("d_k", 64)
        self.d_v: int = params.get("d_v", 64)
        self.positional_encoding_type: str = params.get("positional_encoding", "sinusoidal")
        self.vocab_size: int = params.get("vocab_size", 30000)
        self.max_seq_len: int = params.get("max_seq_len", 5000)

        # Special tokens (defaults if not provided in params)
        self.bos_idx: int = params.get("bos_idx", 2)
        self.eos_idx: int = params.get("eos_idx", 3)

        # Shared token embedding layer. Embeddings are scaled by sqrt(d_model) and are tied with generator.
        self.embedding = nn.Embedding(self.vocab_size, self.d_model)
        self.dropout = nn.Dropout(self.dropout_prob)
        self.positional_encoding = PositionalEncoding(self.d_model, self.dropout_prob,
                                                      max_len=self.max_seq_len,
                                                      mode=self.positional_encoding_type)

        # Encoder stack: a ModuleList of EncoderLayer modules.
        self.encoder_layers_list = nn.ModuleList([
            EncoderLayer(self.d_model, self.num_heads, self.d_ff, self.d_k, self.d_v, self.dropout_prob)
            for _ in range(self.encoder_layers)
        ])

        # Decoder stack: a ModuleList of DecoderLayer modules.
        self.decoder_layers_list = nn.ModuleList([
            DecoderLayer(self.d_model, self.num_heads, self.d_ff, self.d_k, self.d_v, self.dropout_prob)
            for _ in range(self.decoder_layers)
        ])

        # Generator: linear layer to project decoder outputs to vocabulary logits.
        self.generator = nn.Linear(self.d_model, self.vocab_size)
        # Weight tying: share weights between target embedding and generator.
        self.generator.weight = self.embedding.weight

    def _generate_subsequent_mask(self, size: int) -> torch.Tensor:
        """
        Generate a causal mask to ensure that decoder self-attention does not attend to subsequent positions.
        Returns a tensor mask of shape (1, size, size) with True indicating masked positions.
        """
        mask = torch.triu(torch.ones(size, size, device=self.embedding.weight.device, dtype=torch.bool), diagonal=1)
        return mask.unsqueeze(0)

    def _encode(self, src: torch.Tensor) -> torch.Tensor:
        """
        Encode the source sequence.
        Args:
            src: Tensor of shape (batch_size, src_seq_len)
        Returns:
            Encoder output: Tensor of shape (batch_size, src_seq_len, d_model)
        """
        src_emb = self.embedding(src) * math.sqrt(self.d_model)
        src_emb = self.positional_encoding(src_emb)
        enc_output = src_emb
        for layer in self.encoder_layers_list:
            enc_output = layer(enc_output, mask=None)
        return enc_output

    def _decode(self, tgt: torch.Tensor, encoder_output: torch.Tensor) -> torch.Tensor:
        """
        Decode the target sequence using the encoder output as memory.
        Args:
            tgt: Tensor of shape (batch_size, tgt_seq_len)
            encoder_output: Tensor of shape (batch_size, src_seq_len, d_model)
        Returns:
            Decoder output: Tensor of shape (batch_size, tgt_seq_len, d_model)
        """
        tgt_emb = self.embedding(tgt) * math.sqrt(self.d_model)
        tgt_emb = self.positional_encoding(tgt_emb)
        tgt_seq_len = tgt.size(1)
        tgt_mask = self._generate_subsequent_mask(tgt_seq_len)
        dec_output = tgt_emb
        for layer in self.decoder_layers_list:
            dec_output = layer(dec_output, encoder_output, tgt_mask=tgt_mask)
        return dec_output

    def forward(self, x: torch.Tensor, target: Optional[torch.Tensor] = None) -> torch.Tensor:
        """
        Forward pass of the Transformer.
        Args:
            x: Source token tensor of shape (batch_size, src_seq_len)
            target: Optional target token tensor of shape (batch_size, tgt_seq_len).
                    If provided, the model performs full encoder-decoder computation (teacher forcing).
        Returns:
            If target is provided, returns logits of shape (batch_size, tgt_seq_len, vocab_size);
            otherwise, returns the encoder output.
        """
        encoder_output = self._encode(x)
        if target is not None:
            decoder_output = self._decode(target, encoder_output)
            logits = self.generator(decoder_output)
            return logits
        else:
            return encoder_output

    def generate(self, x: torch.Tensor, max_length: int, beam_size: int, length_penalty: float) -> torch.Tensor:
        """
        Performs beam search decoding for inference.
        Args:
            x: Source token tensor of shape (batch_size, src_seq_len). Typically batch_size=1.
            max_length: Maximum output sequence length.
            beam_size: Beam size for beam search.
            length_penalty: Penalty factor to balance length in beam scoring.
        Returns:
            Tensor of shape (batch_size, generated_seq_len) containing generated token indices.
        """
        self.eval()
        device = x.device
        with torch.no_grad():
            encoder_output = self._encode(x)
            batch_size = x.size(0)
            all_outputs = []
            # Process each sample in the batch individually.
            for b in range(batch_size):
                enc_out = encoder_output[b].unsqueeze(0)  # (1, src_seq_len, d_model)
                # Initialize beam with the beginning-of-sequence token.
                beam = [([self.bos_idx], 0.0)]
                completed_sequences = []
                for _ in range(max_length):
                    new_beam = []
                    for seq, score in beam:
                        # If the last token is EOS, carry this sequence forward without expansion.
                        if seq[-1] == self.eos_idx:
                            completed_sequences.append((seq, score))
                            continue
                        tgt_seq = torch.tensor(seq, dtype=torch.long, device=device).unsqueeze(0)  # (1, current_len)
                        dec_out = self._decode(tgt_seq, enc_out)  # (1, current_len, d_model)
                        logits = self.generator(dec_out[:, -1, :])  # (1, vocab_size)
                        log_probs = torch.log_softmax(logits, dim=-1)  # (1, vocab_size)
                        topk_log_probs, topk_indices = torch.topk(log_probs, beam_size, dim=-1)
                        topk_log_probs = topk_log_probs.squeeze(0)
                        topk_indices = topk_indices.squeeze(0)
                        for k in range(beam_size):
                            new_seq = seq + [topk_indices[k].item()]
                            new_score = score + topk_log_probs[k].item()
                            new_beam.append((new_seq, new_score))
                    if not new_beam:
                        break
                    # Apply length penalty: normalized_score = score / (sequence_length^length_penalty)
                    new_beam = sorted(new_beam, key=lambda x: x[1] / (len(x[0]) ** length_penalty), reverse=True)
                    beam = new_beam[:beam_size]
                if completed_sequences:
                    completed_sequences = sorted(completed_sequences,
                                                 key=lambda x: x[1] / (len(x[0]) ** length_penalty),
                                                 reverse=True)
                    best_seq = completed_sequences[0][0]
                else:
                    best_seq = beam[0][0]
                all_outputs.append(torch.tensor(best_seq, dtype=torch.long, device=device))
            # Pad generated sequences within the batch to the same length.
            max_output_len = max(seq.size(0) for seq in all_outputs)
            padded_outputs = []
            pad_idx = self.embedding.padding_idx if hasattr(self.embedding, "padding_idx") and self.embedding.padding_idx is not None else 0
            for seq in all_outputs:
                if seq.size(0) < max_output_len:
                    pad_length = max_output_len - seq.size(0)
                    padding = torch.full((pad_length,), pad_idx, device=device, dtype=torch.long)
                    seq = torch.cat([seq, padding], dim=0)
                padded_outputs.append(seq.unsqueeze(0))
            return torch.cat(padded_outputs, dim=0)
