```python
"""evaluation.py

This module implements the Evaluation class which runs inference on the evaluation dataset using the
trained TransformerModel, applies beam search decoding with parameters read from the configuration, and
computes evaluation metrics. For translation tasks, the BLEU score is computed using sacreBLEU.
For constituency parsing, an F1 score is computed based on bracket spans extracted from the predicted
and reference parse trees.

Required external packages:
    torch==1.9.0
    numpy==1.21.0
    sacrebleu==2.0.0
    tqdm==4.62.0
"""

import torch
from typing import Any, Dict, List, Optional

import sacrebleu
from tqdm import tqdm


class Evaluation:
    """
    Evaluation class for post-training inference and metric computation.
    It utilizes the TransformerModel.generate() method with beam search to produce predictions
    and then computes BLEU (for translation) or F1 (for constituency parsing) scores.
    """

    def __init__(self, model: torch.nn.Module, eval_data: Any, config: Dict[str, Any]) -> None:
        """
        Initializes the Evaluation object with the model, evaluation data, and configuration.

        Args:
            model: An instance of TransformerModel.
            eval_data: Evaluation dataset (typically a DataLoader) containing preprocessed batches.
            config: Dictionary of configurations (from config.yaml) including inference parameters.
        """
        self.model: torch.nn.Module = model
        self.eval_loader: Any = eval_data
        self.config: Dict[str, Any] = config

        # Set the model to evaluation mode (this disables dropout, etc.)
        self.model.eval()

        # Determine if the task is parsing or translation.
        # If the configuration contains a "parsing" inference section, use that; else default to translation.
        if "parsing" in self.config and "inference" in self.config["parsing"]:
            beam_search_cfg: Dict[str, Any] = self.config["parsing"]["inference"].get("beam_search", {})
            self.task: str = "parsing"
        else:
            beam_search_cfg = self.config.get("inference", {}).get("beam_search", {})
            self.task = "translation"

        # Set beam search parameters with default values if not provided.
        if self.task == "translation":
            self.beam_size: int = int(beam_search_cfg.get("beam_size", 4))
            self.length_penalty: float = float(beam_search_cfg.get("length_penalty", 0.6))
            self.max_length_offset: int = int(beam_search_cfg.get("max_length_offset", 50))
        elif self.task == "parsing":
            self.beam_size = int(beam_search_cfg.get("beam_size", 21))
            self.length_penalty = float(beam_search_cfg.get("length_penalty", 0.3))
            self.max_length_offset = int(beam_search_cfg.get("max_length_offset", 300))
        else:
            # Fallback to translation defaults.
            self.beam_size, self.length_penalty, self.max_length_offset = 4, 0.6, 50
            self.task = "translation"

        # Optional: Use a vocabulary mapping (index-to-token) if provided in the configuration.
        # This mapping is used to decode token indices to human-readable strings.
        self.idx2token: Optional[Dict[int, str]] = config.get("idx2token", None)

        # Set indices for special tokens
        self.pad_idx: int = 0  # Default pad token index.
        self.bos_idx: int = getattr(self.model, "bos_idx", 2)
        self.eos_idx: int = getattr(self.model, "eos_idx", 3)

    def _decode_sequence(self, seq: torch.Tensor) -> str:
        """
        Decode a sequence of token indices into a string.
        It removes the beginning-of-sequence token, stops at the end-of-sequence token, and ignores padding.

        Args:
            seq: A 1-D tensor of token indices.

        Returns:
            A string representation of the decoded sequence.
        """
        token_ids: List[int] = seq.tolist()
        decoded_tokens: List[str] = []
        for token_id in token_ids:
            if token_id == self.bos_idx:
                continue
            if token_id == self.eos_idx:
                break
            if token_id == self.pad_idx:
                continue
            if self.idx2token is not None:
                token_str: str = self.idx2token.get(token_id, f"<{token_id}>")
            else:
                # Fallback: simply convert token index to string.
                token_str = str(token_id)
            decoded_tokens.append(token_str)
        return " ".join(decoded_tokens)

    def _extract_brackets(self, tree: str) -> set:
        """
        Extract bracket spans from a linearized constituency parse tree.
        This implementation tokenizes the tree by whitespace and uses a stack-based algorithm
        to record constituent spans.

        Args:
            tree: A string representing the linearized parse tree.

        Returns:
            A set of tuples, each representing a span (start_index, end_index).
        """
        tokens: List[str] = tree.split()
        spans: set = set()
        stack: List[int] = []
        terminal_count: int = 0
        for token in tokens:
            if token == '(':
                # Push the current terminal count as the starting index for a new constituent.
                stack.append(terminal_count)
            elif token == ')':
                if stack:
                    start: int = stack.pop()
                    end: int = terminal_count
                    if end > start:
                        spans.add((start, end))
            else:
                # Treat any token that is not a parenthesis as a terminal word.
                terminal_count += 1
        return spans

    def _compute_f1_from_spans(self, pred_tree: str, ref_tree: str) -> float:
        """
        Compute the F1 score for a predicted and a reference parse tree based on bracket spans.

        Args:
            pred_tree: The predicted parse tree as a string.
            ref_tree: The reference (gold) parse tree as a string.

        Returns:
            The F1 score (float) for the given parse pair.
        """
        pred_spans = self._extract_brackets(pred_tree)
        ref_spans = self._extract_brackets(ref_tree)
        if not pred_spans and not ref_spans:
            return 1.0
        common = pred_spans.intersection(ref_spans)
        precision: float = len(common) / len(pred_spans) if pred_spans else 0.0
        recall: float = len(common) / len(ref_spans) if ref_spans else 0.0
        if precision + recall == 0:
            return 0.0
        f1: float = 2 * precision * recall / (precision + recall)
        return f1

    def evaluate(self) -> Dict[str, float]:
        """
        Evaluates the model on the provided evaluation dataset.

        For translation tasks:
            - Iterates through batches containing "src" (input) and "tgt" (reference).
            - For each sample, computes the max output length as (input length + max_length_offset),
              performs beam search decoding via model.generate(), and decodes the indices to a string.
            - Uses sacreBLEU to compute the corpus-level BLEU score.

        For parsing tasks:
            - Iterates through batches containing "input" as the source.
            - For each sample, decodes the generated parse tree and uses the same decoded input as
              the reference.
            - Computes an F1 score based on bracket spans and averages over the dataset.

        Returns:
            A dictionary of metrics (e.g., {"BLEU": value} or {"F1": value}).
        """
        self.model.eval()
        predictions: List[str] = []
        references: List[str] = []

        with torch.no_grad():
            for batch in tqdm(self.eval_loader, desc="Evaluating", unit="batch"):
                if self.task == "translation":
                    src_batch: Optional[torch.Tensor] = batch.get("src")
                    tgt_batch: Optional[torch.Tensor] = batch.get("tgt")
                    if src_batch is None or tgt_batch is None:
                        continue  # Skip batches that do not have the required keys.
                    src_batch = src_batch.to(next(self.model.parameters()).device)
                    tgt_batch = tgt_batch.to(next(self.model.parameters()).device)
                    batch_size: int = src_batch.size(0)
                    for i in range(batch_size):
                        # Extract individual sample tensors.
                        src_sample = src_batch[i].unsqueeze(0)  # Shape: (1, seq_len)
                        # Calculate actual source length (non-pad tokens).
                        src_length: int = int((src_sample != self.pad_idx).sum().item())
                        max_length: int = src_length + self.max_length_offset
                        generated_output = self.model.generate(src_sample, max_length, self.beam_size, self.length_penalty)
                        # generated_output shape is (1, generated_seq_len)
                        pred_str: str = self._decode_sequence(generated_output[0])
                        predictions.append(pred_str)
                        ref_str: str = self._decode_sequence(tgt_batch[i])
                        references.append(ref_str)
                elif self.task == "parsing":
                    input_batch: Optional[torch.Tensor] = batch.get("input")
                    if input_batch is None:
                        continue
                    input_batch = input_batch.to(next(self.model.parameters()).device)
                    batch_size = input_batch.size(0)
                    for i in range(batch_size):
                        x_sample = input_batch[i].unsqueeze(0)
                        x_length: int = int((x_sample != self.pad_idx).sum().item())
                        max_length: int = x_length + self.max_length_offset
                        generated_output = self.model.generate(x_sample, max_length, self.beam_size, self.length_penalty)
                        pred_str = self._decode_sequence(generated_output[0])
                        predictions.append(pred_str)
                        # For parsing, use the original input as the reference.
                        ref_str = self._decode_sequence(input_batch[i])
                        references.append(ref_str)
                else:
                    raise ValueError(f"Unsupported task type: {self.task}")

        # Compute evaluation metrics.
        if self.task == "translation":
            # Compute BLEU score using sacreBLEU.
            bleu_result = sacrebleu.corpus_bleu(predictions, [references])
            metrics: Dict[str, float] = {"BLEU": bleu_result.score}
        elif self.task == "parsing":
            # Compute average F1 score over all samples.
            f1_scores: List[float] = []
            for pred_tree, ref_tree in zip(predictions, references):
                f1_scores.append(self._compute_f1_from_spans(pred_tree, ref_tree))
            average_f1: float = sum(f1_scores) / len(f1_scores) if f1_scores else 0.0
            metrics = {"F1": average_f1}
        else:
            metrics = {}

        # Log a few sample predictions and references for qualitative assessment.
        sample_log_count: int = 3
        print("\nSample Predictions and References:")
        for idx in range(min(sample_log_count, len(predictions))):
            print(f"Prediction {idx + 1}: {predictions[idx]}")
            print(f"Reference  {idx + 1}: {references[idx]}\n")

        return metrics
```
